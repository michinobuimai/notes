    # Byte Bit 
        10^3 Bytes = kilobyte       kibi KiB = 1024 oder 2^10 bytes
        10^6 Bytes = megabyte       mebi MiB = 2^20 bytes
        10^9 Bytes = gigabyte       gibi GiB = 2^30 bytes
        10^12 Bytes = terabyte      tebi TiB = 2^40 bytes

        byte = 8 bits
        nibble = 4 bits
