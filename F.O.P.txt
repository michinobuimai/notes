                    // klassennamen ist. Der Konstruktor darf keinen Rückgabe wert besitzten!!!

                    // Abrufbar in ´´´main.java´´´ mit ´´´[klassename] [name object] = new [klassenname]([datentyp] [wert])´´´

                        public [klassenname]( [datentyp] [wert1]){

                        }

                    // diese methode kann dann vom object abgerufen werden

                    // ggebene weise zu modifzieren.

                        public [rückgabe datentyp, void wenn keins] [methode]([datentyp] [wert2]){

                            //(irgendwas wird mit [wert2] gemacht oder sogar gar nicht)

                            // Oder es wird einfach ein wert sei es wert1 aus dem object zurrückgegeben.
                            
                        }
